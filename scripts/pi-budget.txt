# Pi budget (#638): the limits scripts/pi-budget.sh checks (CI job "pi budget",
# .github/workflows/pi-budget.yml). Raising one is a visible diff in the PR,
# like scripts/panic-baseline.txt.
# The target is a Pi Zero 2 W: deploy/skrongle.service caps the server at
# MemoryHigh=48M, MemoryMax=64M, TasksMax=32.
#
# Measured 2026-11-09 on the CI runner (ubuntu-latest, linux-x86_64): binary
# 16639304 bytes, idle RSS 21652 KiB, 3 threads. The Pi is aarch64 and its
# numbers differ; CI cannot measure it, so these are a tripwire for growth.

# Idle server (no optional settings, after 60 requests): resident memory in
# KiB (24 MiB = the measured 21652 + about 10 %; checked on Linux only) and
# thread count (the measured 3 is not tightened: a thread is cheap to add
# deliberately, 12 is far below TasksMax=32).
rss_max_kib 24576
threads_max 12

# Stripped release binary, in bytes, per platform (`uname -s`-`uname -m`).
# The binary may grow by binary_slack_percent over its platform's baseline; a
# platform without a baseline (a Mac, the Pi) only reports its size.
# scripts/pi-budget.sh --update rewrites this platform's line.
binary_slack_percent 5
binary_baseline linux-x86_64 16639304
